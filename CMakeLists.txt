cmake_minimum_required(VERSION 3.16)
project(Muscli LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(miniaudio STATIC src/miniaudio/miniaudio.c)
target_link_libraries(miniaudio PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(UNIX AND NOT APPLE)
    target_link_libraries(miniaudio PUBLIC m)
endif()
add_executable(muscli src/main.cpp src/AudioPlayer/AudioPlayer.cpp
    src/TextInput/TextInput.cpp src/Renderer/Renderer.cpp
    src/Menu/Menu.cpp src/Component/Text/Text.cpp)
target_link_libraries(muscli PRIVATE miniaudio)
target_compile_options(muscli PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(audio_test tests/audio_test.cpp src/AudioPlayer/AudioPlayer.cpp)
    target_link_libraries(audio_test PRIVATE miniaudio)
    add_test(NAME audio COMMAND audio_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/test-tone.mp3)
endif()
